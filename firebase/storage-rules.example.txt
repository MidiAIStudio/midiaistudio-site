rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {
    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && firestore.exists(/databases/(default)/documents/users/$(request.auth.uid))
        && firestore.get(/databases/(default)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isImage() {
      return request.resource.contentType.matches('image/.*')
        || request.resource.contentType == 'application/octet-stream';
    }

    function isVideo() {
      return request.resource.contentType.matches('video/.*')
        || request.resource.contentType in [
          'application/octet-stream',
          'application/mp4'
        ];
    }

    // Prefer the repo-root storage.rules for deploy.
    // Guide/Product CMS: signedIn + contentType (not firestore isAdmin in Storage —
    // firestore.get(isAdmin) can fail even when the client UI sees role=admin).
    // Document writes remain admin-only in Firestore.

    match /guide-images/{slug}/{fileName} {
      allow read: if true;
      allow write: if signedIn()
        && request.resource.size < 20 * 1024 * 1024
        && isImage();
    }

    match /guide-videos/{slug}/{fileName} {
      allow read: if true;
      allow create, update: if signedIn()
        && request.resource.size < 80 * 1024 * 1024
        && isVideo();
      allow delete: if signedIn();
    }

    match /product-images/{sectionId}/{fileName} {
      allow read: if true;
      allow write: if signedIn()
        && request.resource.size < 20 * 1024 * 1024
        && isImage();
    }

    match /product-videos/{sectionId}/{fileName} {
      allow read: if true;
      allow create, update: if signedIn()
        && request.resource.size < 80 * 1024 * 1024
        && isVideo();
      allow delete: if signedIn();
    }

    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
